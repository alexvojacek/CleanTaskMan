cmake_minimum_required(VERSION 3.24)
project(CleanTaskMan VERSION 0.7.0 LANGUAGES CXX RC)
add_executable(CleanTaskMan WIN32 src/main.cpp src/telemetry.cpp src/settings.cpp src/process_actions.cpp src/activity.cpp src/process_disk.cpp src/cpu_parking.cpp src/integration.cpp resources/app.rc)
target_compile_features(CleanTaskMan PRIVATE cxx_std_20)
target_compile_definitions(CleanTaskMan PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_link_libraries(CleanTaskMan PRIVATE d2d1 dwrite dwmapi user32 gdi32 comctl32 ole32 shell32 advapi32 psapi powrprof pdh iphlpapi dxgi wtsapi32 setupapi ws2_32 tdh)
if(MSVC)
  target_compile_options(CleanTaskMan PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(CleanTaskMan PRIVATE -Wall -Wextra)
  target_link_options(CleanTaskMan PRIVATE -municode)
endif()
